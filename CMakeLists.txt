cmake_minimum_required(VERSION 3.28)
project(sunjoo_media_cue_filter VERSION 0.5.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(OBS_MAJOR "33" CACHE STRING "32 or 33")
set(OBS_SDK "${WORKSPACE}/sdk/obs33-final/obs-studio-33.0.0" CACHE PATH "OBS source SDK")
set(OBS_IMPORTS "${WORKSPACE}/sdk/lib33-final" CACHE PATH "OBS import libraries")
set(FFMPEG_SDK "${WORKSPACE}/sdk/obs-deps" CACHE PATH "OBS bundled FFmpeg SDK")
set(QT_SDK "${WORKSPACE}/sdk/qt6" CACHE PATH "Matching OBS Qt SDK")
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets PATHS "${QT_SDK}/lib/cmake" NO_DEFAULT_PATH)
file(READ "${OBS_SDK}/libobs/obs-config.h" obs_header)
string(REGEX MATCH "#define LIBOBS_API_MAJOR_VER ([0-9]+)" obs_major_line "${obs_header}")
if(NOT CMAKE_MATCH_1 STREQUAL OBS_MAJOR)
 message(FATAL_ERROR "OBS SDK major mismatch")
endif()
set(OBS_RELEASE_CANDIDATE 0)
set(OBS_BETA 0)
configure_file("${OBS_SDK}/libobs/obsconfig.h.in" "${CMAKE_BINARY_DIR}/generated/obsconfig.h")
configure_file(src/version.rc.in "${CMAKE_BINARY_DIR}/version.rc" @ONLY)
add_library(sunjoo-media-cue-filter MODULE src/filter.cpp src/editor.cpp src/preview-decoder.cpp src/waveform.cpp "${CMAKE_BINARY_DIR}/version.rc")
target_include_directories(sunjoo-media-cue-filter PRIVATE "${OBS_SDK}/libobs" "${OBS_SDK}/frontend/api" "${CMAKE_BINARY_DIR}/generated" "${FFMPEG_SDK}/include")
target_compile_definitions(sunjoo-media-cue-filter PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN FILTER_VERSION="${PROJECT_VERSION}" FILTER_OBS_MAJOR="${OBS_MAJOR}")
target_compile_options(sunjoo-media-cue-filter PRIVATE /utf-8 /W4)
target_link_libraries(sunjoo-media-cue-filter PRIVATE "${OBS_IMPORTS}/obs.lib" "${OBS_IMPORTS}/obs-frontend-api.lib" Qt6::Core Qt6::Gui Qt6::Widgets)
foreach(lib avformat avcodec avutil swscale)
 target_link_libraries(sunjoo-media-cue-filter PRIVATE "${FFMPEG_SDK}/lib/${lib}.lib")
endforeach()
set_target_properties(sunjoo-media-cue-filter PROPERTIES PREFIX "")
add_executable(filter-tests tests/integration.cpp)
target_include_directories(filter-tests PRIVATE "${OBS_SDK}/libobs" "${CMAKE_BINARY_DIR}/generated")
target_compile_definitions(filter-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(filter-tests PRIVATE /utf-8)
target_link_libraries(filter-tests PRIVATE "${OBS_IMPORTS}/obs.lib")
add_executable(stress-tests tests/stress.cpp)
target_include_directories(stress-tests PRIVATE "${OBS_SDK}/libobs" "${CMAKE_BINARY_DIR}/generated")
target_compile_definitions(stress-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(stress-tests PRIVATE /utf-8)
target_link_libraries(stress-tests PRIVATE "${OBS_IMPORTS}/obs.lib" psapi)
add_executable(preview-tests tests/preview.cpp src/preview-decoder.cpp)
target_include_directories(preview-tests PRIVATE src "${FFMPEG_SDK}/include")
target_compile_definitions(preview-tests PRIVATE NOMINMAX)
target_compile_options(preview-tests PRIVATE /utf-8)
foreach(lib avformat avcodec avutil swscale)
 target_link_libraries(preview-tests PRIVATE "${FFMPEG_SDK}/lib/${lib}.lib")
endforeach()
option(FILTER_FRONTEND_TEST "Build isolated UI test helper, never distribute" OFF)
add_executable(waveform-tests tests/waveform.cpp src/waveform.cpp)
target_include_directories(waveform-tests PRIVATE src "${FFMPEG_SDK}/include")
target_compile_definitions(waveform-tests PRIVATE NOMINMAX)
target_compile_options(waveform-tests PRIVATE /utf-8)
foreach(lib avformat avcodec avutil)
 target_link_libraries(waveform-tests PRIVATE "${FFMPEG_SDK}/lib/${lib}.lib")
endforeach()
if(FILTER_FRONTEND_TEST)
 find_package(Qt6 REQUIRED COMPONENTS Core Widgets PATHS "${WORKSPACE}/sdk/qt6/lib/cmake" NO_DEFAULT_PATH)
 add_library(filter-frontend-test MODULE tests/frontend.cpp)
 target_include_directories(filter-frontend-test PRIVATE "${OBS_SDK}/libobs" "${OBS_SDK}/frontend/api" "${CMAKE_BINARY_DIR}/generated")
 target_compile_definitions(filter-frontend-test PRIVATE NOMINMAX TEST_FILE="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/cue-test.mp4" TEST_WEBM_FILE="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/cue-test.webm" TEST_IMAGE="${CMAKE_CURRENT_SOURCE_DIR}/tests/frontend-${OBS_MAJOR}.png")
 target_compile_definitions(filter-frontend-test PRIVATE TEST_EDITOR_FILE="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/editor-test.mp4" TEST_EDITOR_WEBM_FILE="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/editor-test.webm" TEST_EDITOR_IMAGE="${CMAKE_CURRENT_SOURCE_DIR}/tests/editor-${OBS_MAJOR}.png")
 target_compile_definitions(filter-frontend-test PRIVATE TEST_FRACTIONAL_FILE="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/fractional-test.mp4")
 target_compile_options(filter-frontend-test PRIVATE /utf-8)
 target_link_libraries(filter-frontend-test PRIVATE "${OBS_IMPORTS}/obs.lib" "${OBS_IMPORTS}/obs-frontend-api.lib" Qt6::Core Qt6::Widgets)
endif()
if(OBS_MAJOR STREQUAL "33")
 install(TARGETS sunjoo-media-cue-filter LIBRARY DESTINATION plugins/sunjoo-media-cue-filter RUNTIME DESTINATION plugins/sunjoo-media-cue-filter)
 file(WRITE "${CMAKE_BINARY_DIR}/manifest.json" "{\"name\":\"sunjoo-media-cue-filter\",\"id\":\"com.sunjooan.obs.media-cue-filter\",\"display_name\":\"Sunjoo Media Cue Control\",\"version\":\"${PROJECT_VERSION}\",\"authors\":[\"SunjooAn\"],\"description\":\"Visible media preview cue and program playback. SunjooAn ${PROJECT_VERSION}\",\"os_arch\":\"windows-x64\",\"urls\":{}}\n")
 install(FILES "${CMAKE_BINARY_DIR}/manifest.json" DESTINATION plugins/sunjoo-media-cue-filter/data)
else()
 install(TARGETS sunjoo-media-cue-filter LIBRARY DESTINATION obs-plugins/64bit RUNTIME DESTINATION obs-plugins/64bit)
endif()
